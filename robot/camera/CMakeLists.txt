cmake_minimum_required(VERSION 3.10)
project(RSCameraProbe LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This read-only V4L2 probe runs on the verified Jetson")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(RS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio)
add_executable(robot_camera_snapshot ${RS_ROOT}/cpp/apps/robot_camera_snapshot.cpp)
target_include_directories(robot_camera_snapshot PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(robot_camera_snapshot PRIVATE ${OpenCV_LIBS})
target_compile_options(robot_camera_snapshot PRIVATE -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
add_executable(robot_camera_stream ${RS_ROOT}/cpp/apps/robot_camera_stream.cpp)
target_include_directories(robot_camera_stream PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(robot_camera_stream PRIVATE ${OpenCV_LIBS} Threads::Threads)
target_compile_options(robot_camera_stream PRIVATE -Wall -Wextra -Wpedantic)
