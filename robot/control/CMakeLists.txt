cmake_minimum_required(VERSION 3.10)
project(RSControlGateway LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Robot gateway uses the Linux poll reactor")
endif()
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(RS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(robot_control_gateway ${RS_ROOT}/cpp/apps/robot_control_gateway.cpp ${RS_ROOT}/cpp/src/command_dispatcher.cpp ${RS_ROOT}/cpp/src/motion_guard.cpp ${RS_ROOT}/cpp/src/input_mapping.cpp)
target_include_directories(robot_control_gateway PRIVATE ${RS_ROOT}/cpp/include)
target_compile_options(robot_control_gateway PRIVATE -Wall -Wextra -Wpedantic)

add_executable(robot_gamepad_info ${RS_ROOT}/cpp/apps/robot_gamepad_info.cpp)
target_compile_options(robot_gamepad_info PRIVATE -Wall -Wextra -Wpedantic)

add_executable(robot_gamepad_probe ${RS_ROOT}/cpp/apps/robot_gamepad_probe.cpp ${RS_ROOT}/cpp/src/input_mapping.cpp)
target_include_directories(robot_gamepad_probe PRIVATE ${RS_ROOT}/cpp/include)
target_compile_options(robot_gamepad_probe PRIVATE -Wall -Wextra -Wpedantic)

# Local real USB policy runtime. No actuator or ROS dependencies.
add_executable(robot_usb_guard ${RS_ROOT}/cpp/apps/robot_usb_guard.cpp ${RS_ROOT}/cpp/src/keyboard_ingress.cpp ${RS_ROOT}/cpp/src/single_executor.cpp ${RS_ROOT}/cpp/src/linux_gamepad_reader.cpp ${RS_ROOT}/cpp/src/command_dispatcher.cpp ${RS_ROOT}/cpp/src/motion_guard.cpp ${RS_ROOT}/cpp/src/input_mapping.cpp)
target_include_directories(robot_usb_guard PRIVATE ${RS_ROOT}/cpp/include)
target_compile_options(robot_usb_guard PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(robot_usb_guard PRIVATE Threads::Threads)
option(RS_BUILD_ROS_EXECUTOR "Build dormant ROS output adapter" OFF)
if(RS_BUILD_ROS_EXECUTOR)
    find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs)
    add_library(rs_ros_velocity_sink STATIC ${RS_ROOT}/cpp/ros/velocity_sink.cpp)
    target_include_directories(rs_ros_velocity_sink PUBLIC ${RS_ROOT}/cpp/include ${catkin_INCLUDE_DIRS})
    target_link_libraries(rs_ros_velocity_sink PUBLIC ${catkin_LIBRARIES} Threads::Threads)
    target_compile_options(rs_ros_velocity_sink PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(robot_usb_guard PRIVATE AMR_WITH_ROS_EXECUTOR)
    target_include_directories(robot_usb_guard PRIVATE ${catkin_INCLUDE_DIRS})
    target_link_libraries(robot_usb_guard PRIVATE rs_ros_velocity_sink)
endif()
